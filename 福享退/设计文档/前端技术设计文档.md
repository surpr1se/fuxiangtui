## 5. 组件设计规范

### 5.1 组件设计原则

| 原则 | 说明 |
|------|------|
| **可复用** | 组件不依赖具体业务，可在多个页面使用 |
| **可配置** | 通过props传递参数，不同场景灵活配置 |
| **单一职责** | 每个组件只做一件事，保持简单 |
| **良好封装** | 内部状态内聚，对外暴露清晰的接口 |

## 6. 状态管理方案

### 6.1 方案选型：小程序原生全局数据 + 页面级状态

```
┌─────────────────────────────────────────┐
│  全局数据（app.globalData）             │
│  - 用户信息（token、昵称、头像）         │
│  - 系统配置（社平工资等）               │
└─────────────────────────────────────────┘
           │
           ▼
┌─────────────────────────────────────────┐
│  页面级数据（Page.data）                │
│  - 当前页面业务数据                      │
│  - 表单临时数据                         │
└─────────────────────────────────────────�┘
```

---

### 6.2 全局状态管理实现

**app.js**：

```javascript
App({
  globalData: {
    // 用户相关
    userInfo: null,
    token: null,
    isLoggedIn: false,

    // 计算过程临时数据
    calculateFlow: {
      batchNo: null,
      personalInfo: null,
      paymentDetails: [],
      supplementInfo: null
    },

    // 系统配置
    systemConfig: {
      socialAvgWage: {}
    }
  },

  // 设置用户信息
  setUserInfo(userInfo) {
    this.globalData.userInfo = userInfo;
    this.globalData.isLoggedIn = !!userInfo;
  },

  // 清除登录状态
  clearAuth() {
    this.globalData.userInfo = null;
    this.globalData.token = null;
    this.globalData.isLoggedIn = false;
    wx.removeStorageSync('token');
  },

  // 保存计算过程数据
  setCalculateFlowData(data) {
    this.globalData.calculateFlow = {
      ...this.globalData.calculateFlow,
      ...data
    };
  },

  // 清空计算过程
  clearCalculateFlow() {
    this.globalData.calculateFlow = {
      batchNo: null,
      personalInfo: null,
      paymentDetails: [],
      supplementInfo: null
    };
  }
});
```

## 7. API请求封装

### 7.1 请求封装核心代码

**services/request.js**：

```javascript
const config = require('../config/env');

// 基础请求方法
const request = (options) => {
  return new Promise((resolve, reject) => {
    // 显示loading
    if (options.loading !== false) {
      wx.showLoading({ title: '加载中...', mask: true });
    }

    // 获取token
    const token = wx.getStorageSync('token');

    wx.request({
      url: config.baseUrl + options.url,
      method: options.method || 'GET',
      data: options.data || {},
      header: {
        'Content-Type': 'application/json',
        'Authorization': token ? `Bearer ${token}` : '',
        ...options.header
      },
      success: (res) => {
        wx.hideLoading();

        // 统一响应处理
        if (res.statusCode === 200) {
          const { code, message, data } = res.data;

          if (code === 0) {
            resolve(data);
          } else if (code === 401) {
            // Token失效，跳转登录
            wx.showToast({ title: '登录已过期', icon: 'none' });
            setTimeout(() => {
              wx.navigateTo({ url: '/pages/login/login' });
            }, 1500);
            reject(new Error('Unauthorized'));
          } else {
            wx.showToast({ title: message || '请求失败', icon: 'none' });
            reject(new Error(message));
          }
        } else {
          wx.showToast({ title: '网络错误', icon: 'none' });
          reject(new Error('Network Error'));
        }
      },
      fail: (err) => {
        wx.hideLoading();
        wx.showToast({ title: '网络连接失败', icon: 'none' });
        reject(err);
      }
    });
  });
};

// 快捷方法
const http = {
  get: (url, data, options = {}) => request({ url, method: 'GET', data, ...options }),
  post: (url, data, options = {}) => request({ url, method: 'POST', data, ...options }),
  put: (url, data, options = {}) => request({ url, method: 'PUT', data, ...options }),
  delete: (url, data, options = {}) => request({ url, method: 'DELETE', data, ...options }),
  // 文件上传专用
  upload: (url, filePath, formData = {}) => {
    return new Promise((resolve, reject) => {
      wx.showLoading({ title: '上传中...', mask: true });

      wx.uploadFile({
        url: config.baseUrl + url,
        filePath: filePath,
        name: 'file',
        formData: formData,
        header: {
          'Authorization': `Bearer ${wx.getStorageSync('token')}`
        },
        success: (res) => {
          wx.hideLoading();
          const data = JSON.parse(res.data);
          if (data.code === 0) {
            resolve(data.data);
          } else {
            wx.showToast({ title: data.message || '上传失败', icon: 'none' });
            reject(new Error(data.message));
          }
        },
        fail: reject
      });
    });
  }
};

module.exports = http;
```

const http = require('./request');

module.exports = {

## 9. 性能优化方案

### 9.1 页面加载优化

| 优化项 | 具体措施 |
|--------|---------|
| **首屏加载** | 首页只加载核心内容，非关键数据延迟加载 |
| **图片懒加载** | 长列表图片使用 `lazy-load` 属性 |
| **分包加载** | 后续版本功能放在分包中，按需加载 |
| **资源CDN** | 静态图片资源上传CDN，减少包体积 |

### 9.2 长列表优化（缴费明细）

**问题**：缴费明细可能有几百条，直接渲染卡顿

**解决方案**：

1. **虚拟列表**：只渲染可视区域内的元素
2. **分页加载**：每次加载20条，上拉加载更多
3. **年份折叠**：默认只显示最近2年，往年数据折叠显示

```html
<!-- 分页加载示例 -->

  {{hasMore ? '加载更多' : '没有更多了'}}
</view>
```
